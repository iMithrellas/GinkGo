cmake_minimum_required(VERSION 3.19)
project(ginkgo_kws LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SHERPA_SOURCE OR NOT SHERPA_BUILD)
  message(FATAL_ERROR "Set isolated SHERPA_SOURCE and SHERPA_BUILD")
endif()
execute_process(COMMAND git -C "${SHERPA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git -C "${SHERPA_SOURCE}" status --porcelain OUTPUT_VARIABLE dirty COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL "11afbd009a7f8c08f4bcf2fc1b265d0df4670fbf" OR dirty)
  message(FATAL_ERROR "Unmodified pinned sherpa source required")
endif()
file(SHA256 "${SHERPA_BUILD}/lib/libsherpa-onnx-core.a" core_hash)
file(SHA256 "${SHERPA_BUILD}/_deps/onnxruntime-src/lib/libonnxruntime.so" ort_hash)
if(NOT core_hash STREQUAL "953ab9e614f3055af9b746ec1f9705318499b619193442187d24af38ff2da485" OR NOT ort_hash STREQUAL "4b3607aebd1784b26b6f9b20e4bd974c7ab8287043e4d095cb7d2cb40b5e566e")
  message(FATAL_ERROR "Unaudited core/runtime build: requires separate profile review")
endif()
set(deps "${SHERPA_BUILD}/_deps")
add_library(ginkgo-kws SHARED ginkgo_kws.cc)
target_compile_options(ginkgo-kws PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
target_include_directories(ginkgo-kws PRIVATE "${SHERPA_SOURCE}" "${deps}/onnxruntime-src/include" "${deps}/kaldi_decoder-src" "${deps}/kaldifst-src" "${deps}/openfst-src/src/include" "${deps}/json-src/include" "${deps}/eigen-src")
target_link_directories(ginkgo-kws PRIVATE "${SHERPA_BUILD}/lib" "${deps}/onnxruntime-src/lib")
target_link_libraries(ginkgo-kws PRIVATE sherpa-onnx-core kaldi-native-fbank-core kissfft-float kaldi-decoder-core sherpa-onnx-kaldifst-core ssentencepiece_core sherpa-onnx-fstfar sherpa-onnx-fst onnxruntime pthread m dl)
target_link_options(ginkgo-kws PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "LINKER:--exclude-libs,ALL")
set_target_properties(ginkgo-kws PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib" BUILD_RPATH "${SHERPA_BUILD}/lib;${deps}/onnxruntime-src/lib" INSTALL_RPATH "$ORIGIN")
include(CTest)
if(BUILD_TESTING)
  add_executable(ginkgo-kws-guard-test ginkgo_kws_guard_test.cc)
  target_compile_options(ginkgo-kws-guard-test PRIVATE -UNDEBUG)
  get_target_property(includes ginkgo-kws INCLUDE_DIRECTORIES)
  get_target_property(links ginkgo-kws LINK_LIBRARIES)
  target_include_directories(ginkgo-kws-guard-test PRIVATE ${includes})
  target_link_directories(ginkgo-kws-guard-test PRIVATE "${SHERPA_BUILD}/lib" "${deps}/onnxruntime-src/lib")
  target_link_libraries(ginkgo-kws-guard-test PRIVATE ${links})
  add_test(NAME ginkgo-kws-guards COMMAND ginkgo-kws-guard-test)
endif()
